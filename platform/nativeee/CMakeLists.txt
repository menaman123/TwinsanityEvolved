# platform/nativeee -- the decomp's include/platform/*.h implemented over NativeEE (the ONLY seam).
#
#   wasm/        compiled INTO the game module by the top-level CMake (te_cxx.py): Platform::Memory (memory.cpp), the
#                module entry (entry.cpp), the game-facing debug hooks, the WASI stubs. Plus the GENERATED forwarders
#                (build/gen/seam/seam_forwarders_<header>.cpp): Platform::X::Y -> ps2e_x_y import.
#   sdk_shim/    the few PS2SDK headers the game and the kept libc/malloc include (tamtypes.h, kernel.h).
#   host/        host-side (64-bit, links the engine) bodies of the ps2e_* imports, ONE module per header. An owner adds
#                <header>.cpp here defining the strong ps2e_* symbols; until then the generated WEAK stubs
#                (build/gen/seam/seam_stubs.cpp, the te_seam_stubs archive linked LAST) answer and log once.
#                te_host.h is the shared guest-memory helper + the run settings; te_services.h the services registry
#                (Clock, Disc, Files, Pads, Saves, IoHeap) Platform::System constructs in ServiceStartup order.
#                savestate.* is the port's save-state registry + the frame-boundary checkpoint (every service
#                registers a section), padlog.* the pad log (--record-pad/--play-pad/--pad-script).
#                audio_service.* + audio.cpp + sound.cpp (te_audio.h: the light hooks), stream_service.* +
#                stream.cpp, movie_player.* (Platform::Movie over the engine's ps2e_movie decoder + the frame boundary;
#                te::Settings::skip_movies keeps the skip).
#   tests/       one ctest per header over the host bodies (no module, no disc needed except where a fixture ISO is
#                given: TE_TEST_ISO).

# The owners' bodies in one archive, the generated WEAK stubs in their own (te_seam_stubs, linked LAST by the runner):
# a strong ps2e_* body anywhere before it wins regardless of archive member order (stubs and bodies in ONE archive
# let the stubs' member, once pulled for any unimplemented import, satisfy every import and hide an owner's object).
file(GLOB TE_HOST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/host/*.cpp")
# the built-in mods' manifests (mods/<id>/mod.json) embedded as a generated table; an edited manifest reconfigures
file(GLOB TE_MOD_MANIFESTS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/mods/*/mod.json")
set(_te_mods_cpp "// generated by platform/nativeee/CMakeLists.txt from mods/*/mod.json -- do not edit\n#include <cstddef>\nstruct te_mod_manifest { const char* id; const char* json; };\nextern \"C\" const te_mod_manifest te_mod_manifests[];\nextern \"C\" const unsigned te_mod_manifest_count;\nextern \"C\" const te_mod_manifest te_mod_manifests[] = {\n")
set(_te_mods_count 0)
foreach(manifest ${TE_MOD_MANIFESTS})
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${manifest}")
    get_filename_component(_dir "${manifest}" DIRECTORY)
    get_filename_component(_id "${_dir}" NAME)
    file(READ "${manifest}" _json)
    string(APPEND _te_mods_cpp "    {\"${_id}\", R\"TEMOD(${_json})TEMOD\"},\n")
    math(EXPR _te_mods_count "${_te_mods_count} + 1")
endforeach()
string(APPEND _te_mods_cpp "    {nullptr, nullptr}\n};\nextern \"C\" const unsigned te_mod_manifest_count = ${_te_mods_count}u;\n")
file(MAKE_DIRECTORY "${TE_GEN}/mods")
file(WRITE "${TE_GEN}/mods/mods_manifests.cpp.in" "${_te_mods_cpp}")
configure_file("${TE_GEN}/mods/mods_manifests.cpp.in" "${TE_GEN}/mods/mods_manifests.cpp" COPYONLY)
list(APPEND TE_HOST_SOURCES "${TE_GEN}/mods/mods_manifests.cpp")
# the scripted cutscenes the cross button may skip (Platform::System::CutsceneSkippable, host/cutscene_skip.cpp): the table made
# of the cutscene skip judge's fixture (each cutscene's graphs and whether its skip leaves the watched state)
set(TE_CUTSCENE_SKIP_FIXTURE "${CMAKE_SOURCE_DIR}/fixtures/judges/cutscene_skip_pal.json")
add_custom_command(OUTPUT "${TE_GEN}/cutscenes/cutscene_skip_table.cpp"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tools/cutscenes/skip_table.py" --fixture "${TE_CUTSCENE_SKIP_FIXTURE}"
            --out "${TE_GEN}/cutscenes/cutscene_skip_table.cpp"
    DEPENDS "${CMAKE_SOURCE_DIR}/tools/cutscenes/skip_table.py" "${TE_CUTSCENE_SKIP_FIXTURE}"
    COMMENT "skip_table: the cutscenes the cross button may skip")
list(APPEND TE_HOST_SOURCES "${TE_GEN}/cutscenes/cutscene_skip_table.cpp")
add_library(te_platform_host STATIC ${TE_HOST_SOURCES})
target_include_directories(te_platform_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/host" "${CMAKE_CURRENT_SOURCE_DIR}" "${TE_GEN}/seam" "${TE_ENGINE}")
target_compile_options(te_platform_host PRIVATE -Wall -Wextra)
target_compile_definitions(te_platform_host PUBLIC TE_HAVE_SDL_INPUT=${TE_HAVE_SDL_INPUT})
target_link_libraries(te_platform_host PUBLIC nativeee_libs)
add_dependencies(te_platform_host te_seam nativeee)

# The WEAK stubs in their own archive, linked after every owner library: a strong ps2e_* body anywhere before it wins
# regardless of archive member order.
add_library(te_seam_stubs STATIC "${TE_GEN}/seam/seam_stubs.cpp")
target_include_directories(te_seam_stubs PUBLIC "${TE_GEN}/seam")
add_dependencies(te_seam_stubs te_seam)

if(NOT TE_VITA)   # host-run ctests (and te_test_support, which te_pdesc links); nothing of them is packed for the Vita
    add_subdirectory(tests)
endif()
